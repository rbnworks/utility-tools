      const handlePickFolders = async () => {
        setIsPicking(true);
        try {
          // Tauri v2 API: use pickFolders() from @tauri-apps/plugin-dialog
          // Returns Option<string[]> — null if user cancels
          const { pickFolders } = await import("@tauri-apps/plugin-dialog");
          const result = await pickFolders({ multiple: true });
          if (result && result.length > 0) {
            result.forEach(path => {
              if (path && path.trim()) {
                onAdd(path.trim());
              }
            });
          }
        } catch {
          // Tauri not available (dev mode / not in Tauri context) — do nothing
          // User falls back to manual path entry
        } finally {
          setIsPicking(false);
        }
      };
