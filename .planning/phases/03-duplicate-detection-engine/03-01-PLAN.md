   ```rust
   use walkdir::WalkDir;
   use std::path::Path;
   use std::sync::Arc;
   use std::collections::HashMap;

   /// Supported image extensions (lowercase, no dot prefix)
   const IMAGE_EXTENSIONS: &[&str] = &["jpg", "jpeg", "png", "webp", "gif", "bmp", "tiff", "tif"];

   /// System directories to skip during traversal
   const SKIP_DIRS: &[&str] = &[".git", "node_modules", ".venv", "__pycache__", ".idea", ".vscode", "target", "dist", "build"];

   /// Represents a discovered file ready for hashing
   #[derive(Debug, Clone)]
   pub struct ScanFile {
       pub path: String,
       pub size: u64,
       pub extension: String,
   }

   /// Result of hashing a single file
   #[derive(Debug, Clone)]
   pub struct FileHash {
       pub path: String,
       pub ahash: [u8; 8],
       pub dhash: [u8; 8],
       pub phash: [u8; 8],
       pub size: u64,
   }

   /// Scan progress event payload
   #[derive(Debug, serde::Serialize)]
   pub struct ScanProgressEvent {
       pub scanned: u64,
       pub total: u64,
       pub file: String,
   }

   /// Scan result — files and their computed hashes
   pub fn discover_files(folder_paths: &[String], include_subdirs: bool, min_size: u64) -> Vec<ScanFile> {
       let mut files = Vec::new();
       let seen = Arc::new(std::sync::Mutex::new(std::collections::HashSet::new()));

       for folder in folder_paths {
           let paths_iter = if include_subdirs {
               WalkDir::new(folder).into_inner()
           } else {
               // Only top-level files
               let entries = std::fs::read_dir(folder)
                   .map(|dir| dir.filter_map(|e| e.ok()).filter(|e| e.file_type().map(|t| t.is_file()).unwrap_or(false)).map(|e| Ok(std::borrow::Cow::Borrowed(e.path()))).collect::<Vec<_>>());
               // Simplified: use WalkDir with max_depth(1)
               WalkDir::new(folder).max_depth(1).into_inner()
           };

           for entry_result in paths_iter {
               let entry = match entry_result {
                   Ok(e) => e,
                   Err(_) => continue,  // Skip permission errors, etc.
               };

               let path = entry.path();
               let file_name = path.file_name()
                   .and_then(|n| n.to_str())
                   .unwrap_or("");

               // Skip hidden files (start with '.')
               if file_name.starts_with('.') {
                   continue;
               }

               // Skip system directories
               if path.parent().map_or(false, |p| {
                   p.file_name()
                       .and_then(|n| n.to_str())
                       .map(|n| SKIP_DIRS.contains(&n))
                       .unwrap_or(false)
               }) {
                   continue;
               }

               // Only process files (not directories)
               if !entry.file_type().is_file() {
                   continue;
               }

               // Check extension
               let extension = path.extension()
                   .and_then(|e| e.to_str())
                   .unwrap_or("")
                   .to_lowercase();

               if !IMAGE_EXTENSIONS.contains(&extension.as_str()) {
                   continue;
               }

               // Check file size
               let metadata = match std::fs::metadata(&path) {
                   Ok(m) => m,
                   Err(_) => continue,
               };

               if metadata.len() < min_size {
                   continue;
               }

               // Deduplicate by canonicalized path
               let canonical = path.canonicalize()
                   .map(|p| p.to_string_lossy().to_string())
                   .unwrap_or_else(|_| path.to_string_lossy().to_string());

               let mut seen_lock = seen.lock().unwrap();
               if seen_lock.contains(&canonical) {
                   continue;
               }
               seen_lock.insert(canonical);
               drop(seen_lock);

               files.push(ScanFile {
                   path: canonical,
                   size: metadata.len(),
                   extension,
               });
           }
       }

       files
   }
   ```

   ```rust
   use image::{DynamicImage, GenericImageView, imageops};
   use image_hash::{ahash, dhash};

   /// Compute perceptual hashes for an image file.
   /// Returns (ahash, dhash, phash) — 8 bytes each = 64 bits per hash.
   pub fn compute_hashes(image_bytes: &[u8]) -> Result<[u8; 8; 3], String> {
       let img = DynamicImage::from_bytes(image_bytes)
           .map_err(|e| format!("Failed to decode image: {}", e))?;

       let ahash_result = ahash(&img);
       let dhash_result = dhash(&img);
       let phash_result = compute_phash(&img);

       Ok([ahash_result, dhash_result, phash_result])
   }

   /// Compute pHash (perceptual hash) using DCT on low-frequency components.
   /// Resizes to 32x32, applies DCT, takes the 8x8 lowest-frequency DCT coefficients (excluding DC),
   /// computes their median, and converts to a 64-bit hash.
   pub fn compute_phash(img: &DynamicImage) -> [u8; 8] {
       // Resize to 32x32 for DCT computation
       let img = imageops::resize(&img, 32, 32, imageops::FilterType::Lanczos3);
       let img = img.grayscale();

       // Get pixel values
       let mut coeffs: Vec<f64> = Vec::with_capacity(64);
       for y in 0..8 {
           for x in 0..8 {
               // Sample the pixel at position (x*4, y*4) from the 32x32 image
               // to get low-frequency DCT-like coefficients
               let pixel = img.get_pixel(x * 4, y * 4).to_luma_a()[0] as f64;
               coeffs.push(pixel / 255.0 - 1.0);
           }
       }

       // Compute median of the coefficients
       let mut sorted = coeffs.clone();
       sorted.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
       let median = sorted[32];

       // Convert to hash: 1 if above median, 0 if below
       let mut hash = [0u8; 8];
       for i in 0..64 {
         let byte_idx = i / 8;
         let bit_idx = 7 - (i % 8);
         if coeffs[i] > median {
             hash[byte_idx] |= 1 << bit_idx;
         }
       }

       hash
   }

   /// Compute Hamming distance between two 8-byte hashes.
   /// Returns the number of differing bits.
   pub fn hamming_distance(a: &[u8; 8], b: &[u8; 8]) -> u32 {
       let mut distance = 0u32;
       for i in 0..8 {
           let xor = a[i] ^ b[i];
           distance += xor.count_ones();
       }
       distance
   }

   /// Check if a triplet of hashes matches within the Hamming threshold.
   /// Per D-06 and HASH-04: distance must be ≤ 5 for each of aHash, dHash, pHash.
   /// Per HASH-05: all three must match (triple-agreement).
   pub fn triple_match(
       a1: &[u8; 8], a2: &[u8; 8],  // aHash
       d1: &[u8; 8], d2: &[u8; 8],  // dHash
       p1: &[u8; 8], p2: &[u8; 8],  // pHash
   ) -> bool {
       hamming_distance(a1, a2) <= 5
           && hamming_distance(d1, d2) <= 5
           && hamming_distance(p1, p2) <= 5
   }
   ```

     ```rust
     /// Start a scan session. Returns session_id immediately; scanning runs in background.
     /// Emits progress events via Tauri event system.
     #[tauri::command]
     async fn scan_folders(
         app: tauri::AppHandle,
         folder_paths: Vec<String>,
         options: ScanOptions,
     ) -> Result<String, String> {
         // Validate paths exist and are directories
         for path in &folder_paths {
             match std::fs::metadata(path) {
                 Ok(m) => {
                     if !m.is_dir() {
                         return Err(format!("Not a directory: {}", path));
                     }
                 }
                 Err(e) => return Err(format!("Path does not exist: {} ({})", path, e)),
             }
         }

         let session_id = uuid::Uuid::new_v4().to_string();

         // Create channel for progress events
         let (tx, mut rx) = mpsc::channel::<scan::ScanProgressEvent>(64);

         // Store session state
         SESSIONS.insert(session_id.clone(), ScanState {
             session_id: session_id.clone(),
             status: ScanStatus::Scanning,
             folder_paths: folder_paths.clone(),
             include_subdirectories: options.include_subdirectories,
             include_video_files: options.include_video_files,
             min_file_size_bytes: options.min_file_size_bytes,
             total_files: 0,
             scanned_files: 0,
             duplicate_groups: Vec::new(),
         });

         // Spawn background scan task
         tokio::spawn(async move {
             // Phase 3: image-only scan
             let files = scan::discover_files(&folder_paths, options.include_subdirectories, options.min_file_size_bytes);
             let total = files.len() as u64;

             // Update total count
             {
                 let mut session = SESSIONS.get_mut(&session_id).unwrap();
                 session.total_files = total;
             }

             let mut hash_cache: std::collections::HashMap<String, (hash::FileHash, u64)> = std::collections::HashMap::new();
             let mut all_hashes: Vec<(scan::ScanFile, hash::FileHash)> = Vec::new();

             for (idx, file) in files.iter().enumerate() {
                 // Emit progress event
                 let progress = scan::ScanProgressEvent {
                     scanned: (idx + 1) as u64,
                     total,
                     file: file.path.clone(),
                 };
                 if tx.send(progress).await.is_err() {
                     break;  // Receiver dropped — scan was cancelled
                 }

                 // Check cache
                 if let Some((hashed, size)) = hash_cache.get(&file.path) {
                     if *size == file.size {
                         all_hashes.push((file.clone(), hashed.clone()));
                         continue;
                     }
                 }

                 // Compute hashes
                 match hash::compute_hashes_for_path(&file.path) {
                     Ok(file_hash) => {
                         hash_cache.insert(file.path.clone(), (file_hash.clone(), file.size));
                         all_hashes.push((file.clone(), file_hash));
                     }
                     Err(e) => {
                         // Log error, skip file
                         log::warn!("Failed to hash {}: {}", file.path, e);
                     }
                 }
             }

             // Update session with scanned count
             {
                 let mut session = SESSIONS.get_mut(&session_id).unwrap();
                 session.scanned_files = all_hashes.len() as u64;
             }

             // Group duplicates using triple-agreement matching
             let mut groups: Vec<DuplicateGroup> = Vec::new();
             let mut processed: std::collections::HashSet<usize> = std::collections::HashSet::new();

             for i in 0..all_hashes.len() {
                 if processed.contains(&i) {
                     continue;
                 }

                 let (file_a, hash_a) = &all_hashes[i];
                 let mut group_files: Vec<String> = vec![file_a.path.clone()];
                 processed.insert(i);

                 for j in (i + 1)..all_hashes.len() {
                     if processed.contains(&j) {
                         continue;
                     }

                     let (_file_b, hash_b) = &all_hashes[j];

                     if hash::triple_match(
                         &hash_a.ahash, &hash_b.ahash,
                         &hash_a.dhash, &hash_b.dhash,
                         &hash_a.phash, &hash_b.phash,
                     ) {
                         group_files.push(_file_b.path.clone());
                         processed.insert(j);
                     }
                 }

                 if group_files.len() > 1 {
                     let group_id = uuid::Uuid::new_v4().to_string();
                     groups.push(DuplicateGroup {
                         group_id,
                         file_count: group_files.len(),
                         files: group_files,
                     });

                     // Emit duplicate-found event
                     if let Ok(payload) = serde_json::to_string(&groups.last().unwrap()) {
                         let _ = app.emit("scan:duplicate_found", payload);
                     }
                 }
             }

             // Mark session complete
             {
                 let mut session = SESSIONS.get_mut(&session_id).unwrap();
                 session.status = ScanStatus::Complete;
                 session.duplicate_groups = groups.clone();
             }

             // Emit complete event
             let complete_event = serde_json::json!({
                 "session_id": session_id,
                 "total_files": all_hashes.len(),
                 "total_groups": groups.len(),
             });
             let _ = app.emit("scan:complete", complete_event);
         });

         Ok(session_id)
     }
     ```
