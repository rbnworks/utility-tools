   ```rust
   /// Compute hashes for a video file by extracting frames and hashing each one.
   /// Extracts one frame every 5 seconds (per D-07).
   /// Returns vector of (ahash, dhash, phash) triplets — one per extracted frame.
   /// Fallback: if fewer than 3 extractable frames, hashes the first frame only.
   pub fn compute_hashes_for_video(video_path: &str) -> Result<Vec<[u8; 8; 3]>, String> {
       ffmpeg::init().map_err(|e| format!("FFmpeg init failed: {}", e))?;

       let input = ffmpeg::media::Input::open_path(video_path)
           .map_err(|e| format!("Failed to open video {}: {}", video_path, e))?;

       let stream = input.streams().best(ffmpeg::media::Type::Video)
           .ok_or("No video stream found")?;
       let video_stream = stream.as_video().ok_or("Stream is not video")?;

       let fps = video_stream.rate();
       if fps == 0.0 {
           return Err(format!("Video {} has zero frame rate", video_path));
       }

       let ctx = ffmpeg::codec::context::Context::from_parameters(stream.parameters())
           .map_err(|e| format!("Failed to create decoder context: {}", e))?;
       let mut decoder = ctx.decoder().video(video_stream.id())
           .map_err(|e| format!("Failed to create video decoder: {}", e))?;

       // Frame interval at 5-second boundaries using PTS
       let time_base = video_stream.time_base();
       let five_secs = ffmpeg::Rational::new(5, 1);
       let frame_interval = (five_secs.num() as i64 * time_base.den as i64)
           / (five_secs.den as i64 * time_base.num as i64);
       let frame_interval = if frame_interval <= 0 { 5 } else { frame_interval };

       // Scaler: convert source format -> GRAY8, resize to 32x32 for DCT
       let scaler = ffmpeg::sws::Context::get_with_format(
           video_stream.width() as i32,
           video_stream.height() as i32,
           video_stream.format().into(),
           32, 32,
           ffmpeg::format::Pixel::GRAY8,
           ffmpeg::sws::Flags::BILINEAR,
           None, None, None,
       ).map_err(|e| format!("Failed to create scaler: {}", e))?;

       let mut frame_hashes: Vec<[u8; 8; 3]> = Vec::new();
       let mut next_frame_pts = 0i64;

       let mut pkt = ffmpeg::Packet::empty();
       let mut raw_frame = ffmpeg::frame::Video::empty();
       let mut out_frame = ffmpeg::frame::Video::empty();

       for (stream_id, pkt_result) in input.packets() {
           if stream_id != stream.id() { continue; }
           pkt = pkt_result.map_err(|e| format!("Read error: {}", e))?;
           decoder.send_packet(&pkt).ok();

           while decoder.receive_frame(&mut raw_frame).is_ok() {
               let pts = raw_frame.pts().unwrap_or(0);

               if pts >= next_frame_pts {
                   scaler.run(&raw_frame, &mut out_frame).ok();

                   if out_frame.width() == 32 && out_frame.height() == 32 {
                       let data = out_frame.data(0);
                       if data.len() >= 32 * 32 {
                           let pixels: Vec<u8> = data[0..32*32].to_vec();
                           let img = image::GrayImage::from_vec(32, 32, pixels)
                               .map(DynamicImage::ImageLuma8);
                           if let Ok(hashes) = compute_hashes(&img) {
                               frame_hashes.push(hashes);
                           }
                       }
                   }
                   next_frame_pts = pts + frame_interval;
               }

               // Cap at 20 frames to prevent memory issues on long videos
               if frame_hashes.len() >= 20 { break; }
           }
           if frame_hashes.len() >= 20 { break; }
       }

       // D-07 Fallback: try first frame only if no frames extracted
       if frame_hashes.is_empty() {
           let mut pkt2 = ffmpeg::Packet::empty();
           for (stream_id, pkt_result) in input.packets() {
               if stream_id != stream.id() { continue; }
               pkt2 = pkt_result.map_err(|e| format!("Read error: {}", e))?;
               decoder.send_packet(&pkt2).ok();
               if decoder.receive_frame(&mut raw_frame).is_ok() {
                   scaler.run(&raw_frame, &mut out_frame).ok();
                   if out_frame.width() == 32 && out_frame.height() == 32 {
                       let data = out_frame.data(0);
                       if data.len() >= 32 * 32 {
                           let pixels: Vec<u8> = data[0..32*32].to_vec();
                           let img = image::GrayImage::from_vec(32, 32, pixels)
                               .map(DynamicImage::ImageLuma8);
                           if let Ok(hashes) = compute_hashes(&img) {
                               frame_hashes.push(hashes);
                           }
                       }
                   }
                   break;
               }
           }
       }

       if frame_hashes.is_empty() {
           Err(format!("No frames extracted from video: {}", video_path))
       } else {
           Ok(frame_hashes)
       }
   }
   ```
   NOTE: Frame cap at 20 frames. At 5s intervals, a 100s video yields ~20 frames.

3. Add video_duplicate_detection() — frame-set comparison:
   ```rust
   /// Compare two video files for duplication via frame-set comparison.
   /// Per D-07: N >= 3 matching frames AND >= 50% of analyzed frames match.
   pub fn videos_match(frame_hashes_a: &[[u8; 8; 3]], frame_hashes_b: &[[u8; 8; 3]]) -> bool {
       if frame_hashes_a.is_empty() || frame_hashes_b.is_empty() { return false; }
       let num_frames = frame_hashes_a.len().min(frame_hashes_b.len());
       let mut matching: u32 = 0;
       for i in 0..num_frames {
           if triple_match(
               &frame_hashes_a[i].ahash, &frame_hashes_b[i].ahash,
               &frame_hashes_a[i].dhash, &frame_hashes_b[i].dhash,
               &frame_hashes_a[i].phash, &frame_hashes_b[i].phash,
           ) { matching += 1; }
       }
       matching >= 3 && (matching as f64) >= (num_frames as f64 * 0.5)
   }
   ```

4. Add compute_hashes_for_file() — dispatches by FileType:
   ```rust
   /// Compute hashes for a file, dispatching by file type.
   /// For images: single hash triplet via compute_hashes().
   /// For videos: frame-set hash triplet using the median frame's hash.
   pub fn compute_hashes_for_file(
       path: &str, file_type: &crate::scan::FileType
   ) -> Result<crate::scan::FileHash, String> {
       let data = std::fs::read(path)
           .map_err(|e| format!("Failed to read {}: {}", path, e))?;
       let size = data.len() as u64;

       match file_type {
           crate::scan::FileType::Video => {
               let frame_hashes = compute_hashes_for_video(path)?;
               if frame_hashes.is_empty() {
                   return Err(format!("No frames from video: {}", path));
               }
               let median_idx = frame_hashes.len() / 2;
               let m = &frame_hashes[median_idx];
               Ok(crate::scan::FileHash {
                   path: path.to_string(), ahash: m.ahash,
                   dhash: m.dhash, phash: m.phash, size,
               })
           }
           crate::scan::FileType::Image => {
               let [ahash, dhash, phash] = compute_hashes(&data)?;
               Ok(crate::scan::FileHash {
                   path: path.to_string(), ahash, dhash, phash, size,
               })
           }
       }
   }
   ```
   NOTE: Video files use median frame's hash as the canonical hash for grouping. This enables the same triple-agreement matching (used for images) to work for videos.
