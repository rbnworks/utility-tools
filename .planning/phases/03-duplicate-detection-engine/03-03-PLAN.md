     ```rust
     /// Supported image extensions (lowercase, no dot prefix)
     const IMAGE_EXTENSIONS: &[&str] = &["jpg", "jpeg", "png", "webp", "gif", "bmp", "tiff", "tif"];

     /// Supported video extensions (lowercase, no dot prefix)
     const VIDEO_EXTENSIONS: &[&str] = &["mp4", "mov", "avi", "mkv", "webm"];

     /// All supported file types
     const SUPPORTED_EXTENSIONS: &[&str] = &[
         "jpg", "jpeg", "png", "webp", "gif", "bmp", "tiff", "tif",
         "mp4", "mov", "avi", "mkv", "webm",
     ];

     /// Returns true if the extension is a video file
     pub fn is_video_file(extension: &str) -> bool {
         VIDEO_EXTENSIONS.contains(&extension.to_lowercase().as_str())
     }

     /// Returns true if the extension is an image file
     pub fn is_image_file(extension: &str) -> bool {
         IMAGE_EXTENSIONS.contains(&extension.to_lowercase().as_str())
     }
     ```
   - Add FileType enum and extend ScanFile:
     ```rust
     #[derive(Debug, Clone, PartialEq)]
     pub enum FileType {

     #[derive(Debug, Clone)]
     pub struct ScanFile {
         pub path: String,
         pub size: u64,
         pub extension: String,
         pub file_type: FileType,
     }
     ```
